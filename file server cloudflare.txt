// ---------- Helpers Cryptography ----------
const enc = new TextEncoder();
const dec = new TextDecoder();

function base64ToUint8(str) {
  let norm = str.replace(/-/g, "+").replace(/_/g, "/");
  while (norm.length % 4 !== 0) norm += "=";
  const binary = atob(norm);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i);
  return bytes;
}

function uint8ToBase64(bytes) {
  let binary = "";
  for (let i = 0; i < bytes.byteLength; i++) binary += String.fromCharCode(bytes[i]);
  return btoa(binary);
}

function toHex(bytes) {
  return Array.from(new Uint8Array(bytes)).map(b => b.toString(16).padStart(2, "0")).join("");
}

async function sha256Hex(str) {
  const hash = await crypto.subtle.digest("SHA-256", enc.encode(str));
  return toHex(hash);
}

// Constant-time string comparison (chống timing attack cho mật khẩu Admin & chữ ký)
async function timingSafeEqualStr(a, b) {
  if (typeof a !== "string" || typeof b !== "string") return false;
  const ha = new Uint8Array(await crypto.subtle.digest("SHA-256", enc.encode(a)));
  const hb = new Uint8Array(await crypto.subtle.digest("SHA-256", enc.encode(b)));
  let diff = ha.length ^ hb.length;
  for (let i = 0; i < ha.length; i++) diff |= ha[i] ^ hb[i];
  return diff === 0;
}

async function getAesKey(seed) {
  const hash = await crypto.subtle.digest("SHA-256", enc.encode(seed));
  return crypto.subtle.importKey("raw", hash, { name: "AES-GCM" }, false, ["encrypt", "decrypt"]);
}

async function hmacSign(msg, secret) {
  const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const sig = await crypto.subtle.sign("HMAC", key, enc.encode(msg));
  return toHex(sig);
}

// Bộ nhớ RAM V8 Isolate chống Replay Nonce (0 lệnh ghi KV — không bao giờ tốn hạn mức 1.000 KV put/ngày!)
const MEM_NONCES = new Map();
function checkAndRecordNonce(nonce) {
  const now = Date.now();
  if (MEM_NONCES.size > 4000) {
    for (const [k, exp] of MEM_NONCES) {
      if (now > exp) MEM_NONCES.delete(k);
    }
    if (MEM_NONCES.size > 5000) MEM_NONCES.clear();
  }
  const existingExp = MEM_NONCES.get(nonce);
  if (existingExp && now < existingExp) {
    return false;
  }
  MEM_NONCES.set(nonce, now + 360000);
  return true;
}

async function decryptClientPayload(reqBody, env, secret) {
  const { payload, nonce, ts, sig } = reqBody || {};
  if (!payload || !nonce || !ts || !sig) {
    throw new Error("Missing required security envelope");
  }

  // Chống Replay Attack #1: Dung sai thời gian 300s
  const nowSec = Math.floor(Date.now() / 1000);
  const reqSec = parseInt(ts, 10);
  if (isNaN(reqSec) || Math.abs(nowSec - reqSec) > 300) {
    throw new Error("Request expired / Replay attack detected");
  }

  // Kiểm tra chữ ký HMAC-SHA256 (constant-time)
  const expectedSig = await hmacSign(`${payload}:${nonce}:${ts}`, secret);
  if (!(await timingSafeEqualStr(String(sig).toLowerCase(), expectedSig.toLowerCase()))) {
    throw new Error("Invalid cryptographic signature");
  }

  // Chống Replay Attack #2: Nonce 1 lần trong bộ nhớ Edge Isolate (0 tốn KV put!)
  if (!checkAndRecordNonce(String(nonce))) {
    throw new Error("Replay nonce reused");
  }

  // Giải mã AES-256-GCM
  const key = await getAesKey(`${secret}:${nonce}:${ts}`);
  const combined = base64ToUint8(payload);
  if (combined.length < 28) throw new Error("Payload too short");
  const iv = combined.slice(0, 12);
  const ciphertext = combined.slice(12);

  const decrypted = await crypto.subtle.decrypt({ name: "AES-GCM", iv }, key, ciphertext);
  return JSON.parse(dec.decode(decrypted));
}

async function encryptServerResponse(respObj, nonce, ts, secret) {
  const key = await getAesKey(`${secret}:resp:${nonce}:${ts}`);
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const plaintext = enc.encode(JSON.stringify(respObj));
  const encrypted = await crypto.subtle.encrypt({ name: "AES-GCM", iv }, key, plaintext);

  const combined = new Uint8Array(12 + encrypted.byteLength);
  combined.set(iv, 0);
  combined.set(new Uint8Array(encrypted), 12);

  const payload = uint8ToBase64(combined);
  const sig = await hmacSign(`${payload}:${nonce}`, secret);
  return { payload, sig };
}

async function generateSessionToken(key, machineId, expiresAt, secret, extra = {}) {
  const iat = Math.floor(Date.now() / 1000);
  // Session token sống ngắn (2 giờ) để ép app định kỳ xác thực lại với server
  const maxShortExp = iat + 7200;
  const keyExp = expiresAt ? Math.floor(expiresAt / 1000) : maxShortExp;
  const exp = Math.min(maxShortExp, keyExp);
  const rand = toHex(crypto.getRandomValues(new Uint8Array(6)));
  const payloadObj = {
    key,
    mid: machineId,
    iat,
    exp,
    salt: rand,
    ...extra,
  };
  const bodyB64 = uint8ToBase64(enc.encode(JSON.stringify(payloadObj)));
  const sig = await hmacSign(bodyB64, secret);
  return `${bodyB64}.${sig}`;
}

async function generateFeatureToken(key, machineId, feature, secret) {
  const iat = Math.floor(Date.now() / 1000);
  const exp = iat + 300; // Feature permit sống 5 phút
  const payloadObj = {
    key,
    mid: machineId,
    ft: feature,
    iat,
    exp,
    salt: toHex(crypto.getRandomValues(new Uint8Array(4))),
  };
  const bodyB64 = uint8ToBase64(enc.encode(JSON.stringify(payloadObj)));
  const sig = await hmacSign(bodyB64, secret);
  return { token: `${bodyB64}.${sig}`, exp };
}

function getD1(env) {
  const db = env && (env.DB || env.QT_DB || env.LICENSE_DB);
  return (db && typeof db.prepare === "function") ? db : null;
}

async function ensureD1(env) {
  const db = getD1(env);
  if (!db) return null;
  if (!D1_INIT_SET.has(db)) {
    await db.batch([
      db.prepare("CREATE TABLE IF NOT EXISTS licenses (key TEXT PRIMARY KEY, data TEXT NOT NULL, created_at INTEGER DEFAULT 0)"),
      db.prepare("CREATE TABLE IF NOT EXISTS sys_kv (k TEXT PRIMARY KEY, v TEXT NOT NULL, updated_at INTEGER DEFAULT 0)"),
    ]);
    D1_INIT_SET.add(db);
  }
  return db;
}

function isKvQuotaError(err) {
  const msg = String((err && err.message) || err || "").toLowerCase();
  return msg.includes("limit exceeded") || msg.includes("too many");
}

async function sysGetRaw(env, k) {
  const db = await ensureD1(env);
  if (db) {
    const row = await db.prepare("SELECT v FROM sys_kv WHERE k = ?1").bind(k).first();
    if (row && row.v != null) return String(row.v);
  }
  if (env && env.APP_DATA) {
    try {
      const raw = await env.APP_DATA.get(k);
      if (raw != null && db) {
        try {
          await db.prepare("INSERT OR IGNORE INTO sys_kv (k, v, updated_at) VALUES (?1, ?2, ?3)").bind(k, raw, nowMs()).run();
        } catch {}
      }
      return raw;
    } catch {
      return null;
    }
  }
  return null;
}

  if (env && env.APP_DATA) {
    try {
      const opts = ttlSec ? { expirationTtl: ttlSec } : undefined;
      await env.APP_DATA.put(k, valStr, opts);
      return true;
    } catch (err) {
      if (bestEffort) return false;
      if (isKvQuotaError(err)) throw new Error(formatKvLimitHelp());
      throw err;
    }
  }
  return false;
}

async function sysDeleteRaw(env, k) {
  const db = await ensureD1(env);
  if (db) {
    try { await db.prepare("DELETE FROM sys_kv WHERE k = ?1").bind(k).run(); } catch {}
  }
  if (env && env.APP_DATA) {
    try { await env.APP_DATA.delete(k); } catch {}
  }
}

function setMemLicCache(key, data) {
  if (MEM_LIC_CACHE.size > 2500) {
    const firstKey = MEM_LIC_CACHE.keys().next().value;
    if (firstKey) MEM_LIC_CACHE.delete(firstKey);
  }
  MEM_LIC_CACHE.set(key, JSON.parse(JSON.stringify(data)));
}

function buildLicenseMetadata(data) {
  const mids = Array.isArray(data.machine_ids) ? data.machine_ids : (data.machine_id ? [data.machine_id] : []);
  return {
    v: 1,
    email: String(data.email || "").slice(0, 120),
    created_at: data.created_at || 0,
    activated_at: data.activated_at || null,
    expires_at: data.expires_at || null,
    duration_ms: data.duration_ms || 0,
    start_on_activate: !!data.start_on_activate,
    max_devices: Math.max(1, parseInt(data.max_devices, 10) || 1),
    used_devices: mids.length,
    first_mid: mids[0] ? String(mids[0]).slice(0, 16) : null,
    disabled: !!data.disabled,
  };
}

const getLicense = async (env, key) => {
  const db = await ensureD1(env);
  if (db) {
    const row = await db.prepare("SELECT data FROM licenses WHERE key = ?1").bind(key).first();
    if (row && row.data) {
      try {
        const parsed = JSON.parse(row.data);
        if (parsed && typeof parsed === "object") {
          setMemLicCache(key, parsed);
          return parsed;
        }
      } catch {}
    }
  }

  if (MEM_LIC_CACHE.has(key)) {
    return JSON.parse(JSON.stringify(MEM_LIC_CACHE.get(key)));
  }

  if (env && env.APP_DATA) {
    const raw = await env.APP_DATA.get(key);
    if (!raw) return null;
    try {
      const parsed = JSON.parse(raw);
      if (parsed && typeof parsed === "object") {
        setMemLicCache(key, parsed);
        if (db) {
          try {
            await db.prepare("INSERT OR IGNORE INTO licenses (key, data, created_at) VALUES (?1, ?2, ?3)")
              .bind(key, raw, parsed.created_at || 0).run();
          } catch {}
        }
        return parsed;
      }
    } catch {
      return null;
    }
  }
  return null;
};

const putLicense = async (env, key, data, { allowRamFallback = false } = {}) => {

        return;
      }
      if (isKvQuotaError(err)) {
        throw new Error(formatKvLimitHelp());
      }
      throw err;
    }
  }
};

function randomChunk(len = 5) {
  const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
  const bytes = crypto.getRandomValues(new Uint8Array(len));
  let out = "";
  for (let i = 0; i < len; i++) out += chars[bytes[i] % chars.length];
  return out;
}

function makeKey(prefix = "QT") {
  const cleanPrefix = String(prefix || "QT").toUpperCase().replace(/[^A-Z0-9]/g, "").slice(0, 6) || "QT";
  return `${cleanPrefix}-${randomChunk(5)}-${randomChunk(5)}-${randomChunk(5)}`;
}

// ---------- Security Policy & Tamper Audit (Anti-Crack App — Zero Wasteful KV Writes!) ----------
const DEFAULT_SEC_SETTINGS = {
  enforce_binary_hash: false,
  allowed_hashes: [],
  enforce_source_sig: false,
  allowed_source_sigs: [],
};

let MEM_SEC_SETTINGS = null;
let MEM_SEC_SETTINGS_EXP = 0;
let MEM_OBSERVED_LIST = null;
let MEM_ALERTS_LIST = null;
const MEM_ALERT_THROTTLE = new Map();

async function getSecSettings(env) {
  if (MEM_SEC_SETTINGS && nowMs() < MEM_SEC_SETTINGS_EXP) {
    return MEM_SEC_SETTINGS;
  }
  const raw = await sysGetRaw(env, "SEC:SETTINGS");
  let out = { ...DEFAULT_SEC_SETTINGS };
  if (raw) {
    try { out = { ...DEFAULT_SEC_SETTINGS, ...JSON.parse(raw) }; } catch {}
  }
  MEM_SEC_SETTINGS = out;
  MEM_SEC_SETTINGS_EXP = nowMs() + 60000;
  return out;
}

async function getObservedBuilds(env) {
  if (Array.isArray(MEM_OBSERVED_LIST)) return MEM_OBSERVED_LIST;
  const raw = await sysGetRaw(env, "SEC:OBSERVED");
  try { MEM_OBSERVED_LIST = raw ? JSON.parse(raw) : []; } catch { MEM_OBSERVED_LIST = []; }
  if (!Array.isArray(MEM_OBSERVED_LIST)) MEM_OBSERVED_LIST = [];
  return MEM_OBSERVED_LIST;
}

async function recordObservedBuild(env, integrity) {
  if (!integrity || (!integrity.text_hash && !integrity.source_sig)) return;
  const list = await getObservedBuilds(env);
  const textHash = String(integrity.text_hash || "");
  const sourceSig = String(integrity.source_sig || "");
  const bundleId = String(integrity.bundle_id || "");
  const idx = list.findIndex(x => x.text_hash === textHash && x.source_sig === sourceSig);

  if (idx >= 0) {
    // Đã có mã băm build này rồi -> chỉ cập nhật bộ đếm trên RAM, tốn 0 lệnh ghi KV!
    list[idx].last_seen = nowMs();
    list[idx].count = (list[idx].count || 1) + 1;
    return;
  }

  list.unshift({
    text_hash: textHash,
    source_sig: sourceSig,
    bundle_id: bundleId,
    last_seen: nowMs(),
    count: 1,
  });
  if (list.length > 12) list.length = 12;
  await sysPutRaw(env, "SEC:OBSERVED", JSON.stringify(list), { bestEffort: true });
}

async function getSecurityAlerts(env) {
  if (Array.isArray(MEM_ALERTS_LIST)) return MEM_ALERTS_LIST;
  const raw = await sysGetRaw(env, "SEC:ALERTS");
  try { MEM_ALERTS_LIST = raw ? JSON.parse(raw) : []; } catch { MEM_ALERTS_LIST = []; }
  if (!Array.isArray(MEM_ALERTS_LIST)) MEM_ALERTS_LIST = [];
  return MEM_ALERTS_LIST;
}

async function logSecurityAlert(env, req, reason, details = {}) {
  const ip = getClientIP(req);
  const alerts = await getSecurityAlerts(env);
  alerts.unshift({
    ts: nowMs(),
    ip,
    reason,
    ...details,
  });
  if (alerts.length > 60) alerts.length = 60;

  // Giới hạn lưu cảnh báo tối đa 1 lần / 5 phút mỗi IP để không bao giờ tốn hết hạn mức KV put!
  const lastWrite = MEM_ALERT_THROTTLE.get(ip) || 0;
  if (nowMs() - lastWrite > 300000) {
    if (MEM_ALERT_THROTTLE.size > 1000) MEM_ALERT_THROTTLE.clear();
    MEM_ALERT_THROTTLE.set(ip, nowMs());
    await sysPutRaw(env, "SEC:ALERTS", JSON.stringify(alerts), { bestEffort: true });
  }
}

async function verifyAppIntegrity(req, env, body) {
  const integrity = body && typeof body.integrity === "object" ? body.integrity : null;
  const key = (body.license_key || "").trim().toUpperCase();
  const mid = (body.machine_id || "").trim();

  if (!integrity) {
    await logSecurityAlert(env, req, "Thiếu trường integrity (client giả mạo / bản cũ)", { key, mid });
    return { ok: false, cracked: true, error: "Phiên bản ứng dụng không hợp lệ hoặc đã bị chỉnh sửa." };
  }

  // Ghi nhận hash build để Admin có thể ghim trực tiếp trên web
  await recordObservedBuild(env, integrity);

  // Lọc bỏ các thư viện hệ thống hợp lệ do công cụ cài IPA (ESign / Sideloadly / Feather / AltStore / Jailbreak) tự chèn
  const BENIGN_SIDELOAD_LIBS = [
    "substrate", "mobilesubstrate", "substitute", "ellekit", "libhooker",
    "tweakinject", "cynject", "alderis", "sideload", "livecontainer", "altstore"
  ];
  if (integrity.tampered === true) {
    const rawReasons = Array.isArray(integrity.reasons) ? integrity.reasons : ["tamper"];
    const realReasons = rawReasons.filter(r => {
      const lower = String(r).toLowerCase();
      if (lower.startsWith("injected_dylib:")) {
        const libs = lower.replace("injected_dylib:", "").split(",").map(x => x.trim()).filter(Boolean);
        const dangerous = libs.filter(lib => !BENIGN_SIDELOAD_LIBS.some(ok => lib.includes(ok)));
        return dangerous.length > 0;
      }
      return true;
    });
    if (realReasons.length > 0) {
      const reasonStr = realReasons.join(", ");
      await logSecurityAlert(env, req, `Phát hiện công cụ bẻ khóa/can thiệp: ${reasonStr}`, { key, mid, text_hash: integrity.text_hash });
      return { ok: false, cracked: true, error: "Phát hiện công cụ debug / bẻ khóa can thiệp vào ứng dụng!" };
    }
  }

  const settings = await getSecSettings(env);
  let sourceVerified = false;
  if (settings.enforce_source_sig && Array.isArray(settings.allowed_source_sigs) && settings.allowed_source_sigs.length > 0) {
    const s = String(integrity.source_sig || "").toLowerCase();
    const allowed = settings.allowed_source_sigs.map(x => String(x).toLowerCase());
    if (!s || !allowed.includes(s)) {
      await logSecurityAlert(env, req, "Sai chữ ký mã nguồn (Source đã bị chỉnh sửa)", { key, mid, source_sig: s });
      return { ok: false, cracked: true, error: "Mã nguồn ứng dụng đã bị thay đổi trái phép!" };
    }
    sourceVerified = true;
  }

  // Lưu ý quan trọng trên iOS Sideload: Khi khách cài IPA qua ESign / Scarlet / Feather / Sideloadly / Cert riêng,
  // mã __TEXT sẽ tự đổi khác nhau trên từng máy dù cùng 1 file IPA gốc, trong khi Source Signature luôn giống nhau 100%!
  // Do đó nếu chữ ký Mã Nguồn (source_sig) đã khớp chính chủ thì cho phép máy thứ 2, thứ 3... hoạt động bình thường!
  if (settings.enforce_binary_hash && !sourceVerified && Array.isArray(settings.allowed_hashes) && settings.allowed_hashes.length > 0) {
    const h = String(integrity.text_hash || "").toLowerCase();
    const allowed = settings.allowed_hashes.map(x => String(x).toLowerCase());
    if (!h || !allowed.includes(h)) {
      await logSecurityAlert(env, req, "Sai mã băm __TEXT và chưa xác thực Source Signature", { key, mid, text_hash: h });
      return { ok: false, cracked: true, error: "File thực thi của ứng dụng không hợp lệ hoặc đã bị chỉnh sửa!" };
    }
  }

  if (settings.enforce_anti_proxy && integrity.proxy_active === true) {
    const proxyDetail = String(integrity.proxy_detail || "");
    const isLocalVpnTunnel = proxyDetail.startsWith("127.0.0.1:") || proxyDetail.startsWith("localhost:") || proxyDetail.startsWith("::1:");
    if (!isLocalVpnTunnel) {
      await logSecurityAlert(env, req, `Phát hiện Proxy bắt gói tin MITM ngoài LAN (${proxyDetail || "HTTP Proxy"})`, { key, mid });
      return { ok: false, cracked: false, error: "Vui lòng tắt HTTP Proxy bắt gói tin trên mạng Wi-Fi trước khi sử dụng!" };
    }
  }

  return { ok: true };
}

// ---------- Lớp Bảo Mật 4 Cho App: Khóa Phiên Động (Ephemeral Key) & Chữ Ký X-QT-Proof ----------
async function deriveEphemeralKey(key, machineId, sessionToken, secret) {
  return await hmacSign(`EPHEMERAL_V4:${String(key).toUpperCase()}:${machineId}:${sessionToken}`, secret);
}

async function verifyAppLayer4Proof(req, env, body, nonce, ts, secret) {
  const key = (body.license_key || "").trim().toUpperCase();
  const mid = (body.machine_id || "").trim();
  const sTok = String(body.session_token || "").trim();
  const clientProof = String(req.headers.get("X-QT-Proof") || "").trim();

  if (!sTok || !sTok.includes(".") || !clientProof) {
    await logSecurityAlert(env, req, "Thiếu chứng chỉ phiên Lớp 4 (session_token / X-QT-Proof)", { key, mid });
    return { ok: false, error: "Phiên bảo mật Lớp 4 chưa khởi tạo hoặc đã hết hạn. Vui lòng mở lại ứng dụng." };
  }

  const [pB64, pSig] = sTok.split(".");
  const expectedTokSig = await hmacSign(pB64, secret);
  if (!(await timingSafeEqualStr(pSig.toLowerCase(), expectedTokSig.toLowerCase()))) {
    await logSecurityAlert(env, req, "Giả mạo chữ ký session_token Lớp 4", { key, mid });
    return { ok: false, cracked: true, error: "Chứng chỉ phiên làm việc bị can thiệp!" };
  }

  let tokObj;
  try {
    tokObj = JSON.parse(dec.decode(base64ToUint8(pB64)));
  } catch {
    return { ok: false, error: "Định dạng phiên làm việc không hợp lệ." };
  }

  if (tokObj.mid !== mid) {
    await logSecurityAlert(env, req, "Lệch thiết bị trong session_token Lớp 4", { key, mid });
    return { ok: false, cracked: true, error: "Phiên bảo mật không khớp thiết bị!" };
  }

  const expMs = tokObj.exp > 100000000000 ? tokObj.exp : tokObj.exp * 1000;
  if (nowMs() > expMs) {
    return { ok: false, error: "Phiên làm việc đã hết hạn, vui lòng thử lại." };
  }

  const pathname = new URL(req.url).pathname;
  const ephKey = await deriveEphemeralKey(key, mid, sTok, secret);
  const expectedProof = await hmacSign(`${pathname}:${nonce}:${ts}:${mid}`, ephKey);
  if (!(await timingSafeEqualStr(clientProof.toLowerCase(), expectedProof.toLowerCase()))) {
    await logSecurityAlert(env, req, "Sai chữ ký động X-QT-Proof Lớp 4", { key, mid });
    return { ok: false, error: "Xác thực chữ ký động Lớp 4 thất bại." };
  }

  return { ok: true };
}

// ---------- Server-Side Anti-Crack Admin Session Auth ----------
function parseCookies(req) {
  const header = req.headers.get("Cookie") || "";
  const map = {};
  for (const part of header.split(";")) {
    const eq = part.indexOf("=");
    if (eq > 0) {
      map[part.slice(0, eq).trim()] = decodeURIComponent(part.slice(eq + 1).trim());
    }
  }
  return map;
}

async function getAdminSigningSecret(env) {
  if (env.ADMIN_SECRET_HASH && String(env.ADMIN_SECRET_HASH).trim().length >= 32) {
    return String(env.ADMIN_SECRET_HASH).trim().toLowerCase();
  }
  if (env.ADMIN_SECRET && String(env.ADMIN_SECRET).length > 0) {
    return await sha256Hex(String(env.ADMIN_SECRET));
  }
  return null;
}

async function verifyAdminPassword(pass, env) {
  if (typeof pass !== "string" || !pass) return false;
  const expectedHash = await getAdminSigningSecret(env);
  if (!expectedHash) return false;
  const inputHash = await sha256Hex(pass);
  return await timingSafeEqualStr(inputHash, expectedHash);
}

async function createAdminCookie(req, env) {
  const signKey = await getAdminSigningSecret(env);
  const ipHash = (await sha256Hex(getClientIP(req))).slice(0, 16);
  const uaHash = (await sha256Hex(req.headers.get("User-Agent") || "")).slice(0, 16);
  const csrf = toHex(crypto.getRandomValues(new Uint8Array(16)));
  const exp = Math.floor(Date.now() / 1000) + 7200; // Phiên Admin sống 2 giờ
  const payloadObj = { iph: ipHash, uah: uaHash, csrf, exp, nonce: toHex(crypto.getRandomValues(new Uint8Array(8))) };
  const bodyB64 = uint8ToBase64(enc.encode(JSON.stringify(payloadObj)));
  const sig = await hmacSign(`ADMIN_V3:${bodyB64}`, signKey);
  const token = `${bodyB64}.${sig}`;
  const isHttps = new URL(req.url).protocol === "https:";
  const cookie = `QT_ADMIN_SID=${encodeURIComponent(token)}; Path=/; HttpOnly; SameSite=Strict; Max-Age=7200${isHttps ? "; Secure" : ""}`;
  return { cookie, csrf, exp };
}

async function verifyAdminSession(req, env, requireCsrf = true) {
  const signKey = await getAdminSigningSecret(env);
  if (!signKey) return { ok: false, error: "Server chưa cấu hình ADMIN_SECRET_HASH hoặc ADMIN_SECRET" };

  const cookies = parseCookies(req);
  const token = cookies["QT_ADMIN_SID"];
  if (!token || !token.includes(".")) {
    return { ok: false, error: "Phiên đăng nhập chưa xác thực hoặc đã hết hạn." };
  }

  const [bodyB64, sig] = token.split(".");
  if (!bodyB64 || !sig) return { ok: false, error: "Cookie không hợp lệ." };

  const expectedSig = await hmacSign(`ADMIN_V3:${bodyB64}`, signKey);
  if (!(await timingSafeEqualStr(sig.toLowerCase(), expectedSig.toLowerCase()))) {
    return { ok: false, error: "Chữ ký phiên quản trị bị giả mạo." };
  }

  let payload;
  try {
    payload = JSON.parse(dec.decode(base64ToUint8(bodyB64)));
  } catch {
    return { ok: false, error: "Cookie lỗi định dạng." };
  }

  if (!payload.exp || Math.floor(Date.now() / 1000) > payload.exp) {
    return { ok: false, error: "Phiên quản trị đã hết hạn, vui lòng đăng nhập lại." };
  }

  // Không chặn đổi IPv4 <-> IPv6 động của nhà mạng VNPT/Viettel/FPT để tránh rớt phiên khi tải Danh Sách Key
  const curUaHash = (await sha256Hex(req.headers.get("User-Agent") || "")).slice(0, 16);
  if (payload.uah !== curUaHash) {
    return { ok: false, error: "Trình duyệt không khớp phiên đăng nhập." };
  }

  if (requireCsrf) {
    const hdrCsrf = req.headers.get("X-Admin-CSRF") || "";
    if (!(await timingSafeEqualStr(hdrCsrf, payload.csrf || ""))) {
      return { ok: false, error: "Thiếu hoặc sai mã CSRF bảo mật." };
    }
  }

  return { ok: true, csrf: payload.csrf, exp: payload.exp };
}

// Challenge-Response ký HMAC không trạng thái (0 lệnh ghi KV — đăng nhập Admin mượt ngay cả khi KV hết lượt ghi!)
const MEM_USED_CHAL = new Map();
const MEM_ADMIN_RL = new Map();

async function handleAdminChallenge(req, env) {
  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);
  const signKey = (await getAdminSigningSecret(env)) || SERVER_MASTER_SECRET;
  const ts = Math.floor(Date.now() / 1000);
  const rand = toHex(crypto.getRandomValues(new Uint8Array(12)));
  const sig = await hmacSign(`ADM_CHAL_V4:${ts}:${rand}`, signKey);
  const challenge = `${ts}.${rand}.${sig}`;
  const pinRequired = !!(env.ADMIN_PIN && String(env.ADMIN_PIN).trim().length > 0);
  return json({ ok: true, challenge, pin_required: pinRequired });
}

async function verifyStatelessAdminChallenge(challengeStr, signKey) {
  const parts = String(challengeStr || "").split(".");
  if (parts.length !== 3) return false;
  const [tsStr, rand, sig] = parts;
  const ts = parseInt(tsStr, 10);
  const nowSec = Math.floor(Date.now() / 1000);
  if (isNaN(ts) || Math.abs(nowSec - ts) > 120) return false;
  const expectedSig = await hmacSign(`ADM_CHAL_V4:${ts}:${rand}`, signKey);
  if (!(await timingSafeEqualStr(sig.toLowerCase(), expectedSig.toLowerCase()))) return false;
  if (MEM_USED_CHAL.has(rand)) return false;
  if (MEM_USED_CHAL.size > 1000) MEM_USED_CHAL.clear();
  MEM_USED_CHAL.set(rand, Date.now() + 130000);
  return true;
}

async function handleAdminAuth(req, env) {
  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);
  const expectedHash = await getAdminSigningSecret(env);
  if (!expectedHash) return json({ error: "Chưa cấu hình ADMIN_SECRET_HASH hoặc ADMIN_SECRET trên Cloudflare Worker" }, 500);

  const ip = getClientIP(req);
  let rl = MEM_ADMIN_RL.get(ip) || { fails: 0, lockedUntil: 0 };

  if (rl.lockedUntil && nowMs() < rl.lockedUntil) {
    const remainSec = Math.ceil((rl.lockedUntil - nowMs()) / 1000);
    return json({ ok: false, error: `Nhập sai quá nhiều lần! Khóa IP thêm ${remainSec} giây.` }, 429);
  }

  const body = await req.json().catch(() => ({}));
  let valid = false;

  if (body.challenge && body.proof) {
    const chalOk = await verifyStatelessAdminChallenge(String(body.challenge), expectedHash);
    if (chalOk) {
      const expectedProof = await hmacSign(String(body.challenge), expectedHash);
      valid = await timingSafeEqualStr(String(body.proof).toLowerCase(), expectedProof.toLowerCase());
    }
  } else if (body.password) {
    valid = await verifyAdminPassword(String(body.password), env);
  }

  // Kiểm tra thêm Mã PIN Lớp 2 nếu Admin đã đặt biến ADMIN_PIN trên Cloudflare
  if (valid && env.ADMIN_PIN && String(env.ADMIN_PIN).trim().length > 0) {
    const pinInput = String(body.pin || "").trim();
    const expectedPin = String(env.ADMIN_PIN).trim();
    valid = await timingSafeEqualStr(pinInput, expectedPin);
  }

  if (!valid) {
    // Chống brute-force: tăng đếm lỗi trên RAM & delay nhẹ (0 tốn KV put!)
    rl.fails = (rl.fails || 0) + 1;
    if (rl.fails >= 5) {
      rl.lockedUntil = nowMs() + 15 * 60 * 1000; // Khóa 15 phút
      await logSecurityAlert(env, req, `Brute-force Admin bị chặn (${rl.fails} lần sai)`);
    }
    if (MEM_ADMIN_RL.size > 1000) MEM_ADMIN_RL.clear();
    MEM_ADMIN_RL.set(ip, rl);
    await new Promise(r => setTimeout(r, 450));
    const left = Math.max(0, 5 - rl.fails);
    return json({
      ok: false,
      error: left > 0 ? `Sai mật khẩu quản trị! Còn ${left} lần thử.` : "Sai quá 5 lần! IP đã bị khóa 15 phút."
    }, 401);
  }

  // Đăng nhập thành công -> xóa bộ đếm lỗi trên RAM & cấp HttpOnly Cookie ký HMAC
  MEM_ADMIN_RL.delete(ip);
  const { cookie, csrf, exp } = await createAdminCookie(req, env);
  return json(
    { ok: true, csrf, exp },
    200,
    { "Set-Cookie": cookie }
  );
}

async function handleGenerateKey(req, env) {
  const guard = await requireAdmin(req, env);
  if (guard.res) return guard.res;

  const body = await req.json().catch(() => ({}));
  const days = Math.max(0, parseInt(body.days, 10) || 0);
  const hours = Math.max(0, parseInt(body.hours, 10) || 0);
  const minutes = Math.max(0, parseInt(body.minutes, 10) || 0);
  const quantity = Math.min(100, Math.max(1, parseInt(body.quantity, 10) || 1));
  const maxDevices = Math.min(50, Math.max(1, parseInt(body.max_devices, 10) || 1));
  const prefix = String(body.prefix || "QT").trim();
  const email = String(body.email || "").trim();
  const startOnActivate = body.start_on_activate === true;

  const durationMs = ((days * 24 + hours) * 60 + minutes) * 60 * 1000;
  if (durationMs <= 0) {
    return json({ valid: false, error: "Thời hạn key phải lớn hơn 0." }, 400);
  }

  const labelParts = [];
  if (days > 0) labelParts.push(`${days} ngày`);
  if (hours > 0) labelParts.push(`${hours} giờ`);
  if (minutes > 0) labelParts.push(`${minutes} phút`);
  const durationLabel = labelParts.join(" ");

  const created = nowMs();
  const keys = [];
  for (let i = 0; i < quantity; i++) {
    const key = makeKey(prefix);
    const licData = {
      email: quantity > 1 && email ? `${email} #${i + 1}` : email,
      created_at: created,
      duration_ms: durationMs,
      start_on_activate: startOnActivate,
      activated_at: startOnActivate ? null : created,
      expires_at: startOnActivate ? null : created + durationMs,
      max_devices: maxDevices,
      machine_ids: [],
      machine_id: null,
      disabled: false,
    };
    await putLicense(env, key, licData);
    keys.push(key);
  }

  return json({
    valid: true,
    key: keys[0],
    keys,
    quantity: keys.length,
    max_devices: maxDevices,
    duration_ms: durationMs,
    duration_label: durationLabel,
    start_on_activate: startOnActivate,
  });
}

function licenseObjToListItem(name, lic) {
  const waiting = !!(lic.start_on_activate && !lic.activated_at && !lic.expires_at);
  const mids = Array.isArray(lic.machine_ids) ? lic.machine_ids : (lic.machine_id ? [lic.machine_id] : []);
  return {
    key: name,
    email: lic.email || "",
    created_at: lic.created_at || 0,
    activated_at: lic.activated_at || null,
    expires_at: lic.expires_at || null,
    duration_ms: lic.duration_ms || 0,
    waiting_activation: waiting,
    max_devices: Math.max(1, parseInt(lic.max_devices, 10) || 1),
    machine_ids: mids,
    machine_id: lic.machine_id || mids[0] || null,
    disabled: !!lic.disabled,
  };
}

async function handleListKeys(req, env) {
  const guard = await requireAdmin(req, env);
  if (guard.res) return guard.res;

  const byKey = new Map();
  const db = await ensureD1(env);

  // 1. Đọc từ D1 SQLite (nếu đã gắn DB — 1 truy vấn lấy tới 5.000 key, 0 tốn KV!)
  if (db) {
    const { results } = await db.prepare("SELECT key, data FROM licenses ORDER BY created_at DESC LIMIT 5000").all();
    for (const row of (results || [])) {
      try {
        const lic = JSON.parse(row.data);
        if (lic && typeof lic === "object") {
          byKey.set(row.key, licenseObjToListItem(row.key, lic));
        }
      } catch {}
    }
  }

  // 2. Đồng bộ thêm từ RAM Cache (phòng trường hợp vừa kích hoạt khi KV chạm trần put)
  for (const [k, lic] of MEM_LIC_CACHE) {
    if (!byKey.has(k) && lic && typeof lic === "object") {
      byKey.set(k, licenseObjToListItem(k, lic));
    }
  }

  // 3. Quét thêm KV APP_DATA để không sót bất kỳ key cũ nào (an toàn ngay cả khi KV hết lượt put!)
  let moreLegacy = false;
  if (env && env.APP_DATA) {
    const INTERNAL_PREFIXES = ["~NONCE:", "NONCE:", "ADMIN_RL:", "SEC:", "IMG:", "ADM_CHAL:"];
    const allEntries = [];
    let cursor = undefined;
    let listPages = 0;

    try {
      while (listPages < 5) {
        listPages++;
        const listed = await env.APP_DATA.list({ limit: 1000, cursor });
        for (const entry of listed.keys) {
          const name = entry.name;
          if (byKey.has(name)) continue;
          if (INTERNAL_PREFIXES.some(p => name.startsWith(p))) continue;
          allEntries.push(entry);
        }
        if (listed.list_complete || !listed.cursor) break;
        cursor = listed.cursor;
      }
    } catch {}

    const legacyEntries = [];
    for (const entry of allEntries) {
      const name = entry.name;
      const meta = entry.metadata;
      if (meta && typeof meta === "object" && meta.v === 1) {
        const waiting = !!(meta.start_on_activate && !meta.activated_at && !meta.expires_at);
        byKey.set(name, {
          key: name,
          email: meta.email || "",
          created_at: meta.created_at || 0,
          activated_at: meta.activated_at || null,
          expires_at: meta.expires_at || null,
          duration_ms: meta.duration_ms || 0,
          waiting_activation: waiting,
          max_devices: Math.max(1, parseInt(meta.max_devices, 10) || 1),
          machine_ids: Array.from({ length: meta.used_devices || 0 }, (_, idx) => idx === 0 ? (meta.first_mid || "bound") : "bound"),
          machine_id: meta.first_mid || null,
          disabled: !!meta.disabled,
        });
      } else {
        legacyEntries.push(entry);
      }
    }

    for (let i = legacyEntries.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      const tmp = legacyEntries[i];
      legacyEntries[i] = legacyEntries[j];
      legacyEntries[j] = tmp;
    }

    let kvWriteBlocked = false;
    let subreqBudget = 36; // Dư an toàn dưới mốc 50 subrequests của Cloudflare Free
    let processedLegacy = 0;

    for (const entry of legacyEntries) {
      if (subreqBudget < (kvWriteBlocked ? 1 : 2)) break;
      processedLegacy++;
      const name = entry.name;
      subreqBudget--;
      const lic = await getLicense(env, name); // getLicense cũng tự động lưu vào D1 nếu đã gắn DB!
      if (!lic || typeof lic !== "object") continue;

      if (!db && !kvWriteBlocked) {
        try {
          subreqBudget--;
          await putLicense(env, name, lic);
        } catch {
          // Nếu KV đang chạm trần 1.000 put/ngày -> dừng thử put tiếp để đọc hết key bằng get()!
          kvWriteBlocked = true;
        }
      }
      byKey.set(name, licenseObjToListItem(name, lic));
    }

    moreLegacy = !kvWriteBlocked && processedLegacy < legacyEntries.length;
  }

  const items = Array.from(byKey.values());
  items.sort((a, b) => (b.created_at || 0) - (a.created_at || 0));
  return json({ ok: true, items, more_legacy: moreLegacy, storage: db ? "d1" : "kv" });
}

async function handleKeyInfo(req, env) {
  const guard = await requireAdmin(req, env);
  if (guard.res) return guard.res;

  const body = await req.json().catch(() => ({}));
  const key = (body.key || "").trim().toUpperCase();
  const lic = await getLicense(env, key);
  if (!lic) return json({ found: false, error: "Key không tồn tại" });

  return json({
    found: true,
    key,
    email: lic.email,
    created_at: lic.created_at,
    activated_at: lic.activated_at || null,
    expires_at: lic.expires_at,
    duration_ms: lic.duration_ms || 0,
    expired: lic.expires_at ? nowMs() > lic.expires_at : false,
    machine_id: lic.machine_id,
    disabled: !!lic.disabled,
  });
}

async function handleExtendKey(req, env) {
  const guard = await requireAdmin(req, env);
  if (guard.res) return guard.res;

  const body = await req.json().catch(() => ({}));
  const key = (body.key || "").trim().toUpperCase();
  const addHours = parseFloat(body.hours);
  if (!key || isNaN(addHours) || addHours <= 0) return json({ error: "Dữ liệu không hợp lệ" }, 400);

  const lic = await getLicense(env, key);
  if (!lic) return json({ error: "Key không tồn tại" }, 404);

  const addMs = Math.round(addHours * 3600 * 1000);
  if (lic.start_on_activate && !lic.activated_at && !lic.expires_at) {
    lic.duration_ms = (lic.duration_ms || 0) + addMs;
  } else {
    const base = Math.max(nowMs(), lic.expires_at || nowMs());
    lic.expires_at = base + addMs;
  }
  await putLicense(env, key, lic);
  return json({ success: true, expires_at: lic.expires_at, duration_ms: lic.duration_ms });
}

// ---------- Core License Evaluation Helper ----------
async function evaluateLicenseForApp(req, env, body, allowBindHwid = true) {
  const integ = await verifyAppIntegrity(req, env, body);
  if (!integ.ok) {
    return { valid: false, cracked: integ.cracked !== false, error: integ.error };
  }

  const key = (body.license_key || "").trim().toUpperCase();
  const machineId = (body.machine_id || "").trim();
  if (!key || !machineId) {
    return { valid: false, error: "Thiếu license key hoặc mã thiết bị." };
  }

  const lic = await getLicense(env, key);
  if (!lic) {
    return { valid: false, error: "License key không tồn tại hoặc đã bị xóa." };
  }
  if (lic.disabled) {
    return { valid: false, cracked: true, error: "License key đã bị khóa bởi quản trị viên." };
  }

  // Hỗ trợ key tính giờ từ lúc kích hoạt lần đầu
  let dirty = false;
  if (lic.start_on_activate && !lic.activated_at && lic.duration_ms > 0) {
    const now = nowMs();
    lic.activated_at = now;
    lic.expires_at = now + lic.duration_ms;
    dirty = true;
  }

  if (lic.expires_at && nowMs() > lic.expires_at) {
    return { valid: false, expired: true, expires_at: lic.expires_at, error: "Key đã hết hạn sử dụng." };
  }

  const mids = Array.isArray(lic.machine_ids) ? lic.machine_ids.slice() : (lic.machine_id ? [lic.machine_id] : []);
  const maxDevices = Math.max(1, parseInt(lic.max_devices, 10) || 1);

  if (!mids.includes(machineId)) {
    if (mids.length >= maxDevices) {
      return {
        valid: false,
        cracked: false,
        error: `Key này chỉ cho phép ${maxDevices} thiết bị (đã gắn đủ ${mids.length}/${maxDevices} máy). Vào Admin bấm 'Reset Máy' hoặc tạo key nhiều thiết bị!`,
      };
    }
    if (!allowBindHwid) {
      return { valid: false, error: "Thiết bị chưa được kích hoạt với key này." };
    }
    mids.push(machineId);
    lic.machine_ids = mids;
    lic.machine_id = mids[0];
    dirty = true;
  }

  if (dirty) {
    await putLicense(env, key, lic, { allowRamFallback: true });
  }

  const sessionToken = await generateSessionToken(key, machineId, lic.expires_at, SERVER_MASTER_SECRET, { v: CONFIG_VERSION });
  const ephemeralKey = await deriveEphemeralKey(key, machineId, sessionToken, SERVER_MASTER_SECRET);
  return {
    valid: true,
    key,
    machineId,
    created_at: lic.activated_at || lic.created_at || nowMs(),
    expires_at: lic.expires_at,
    session_token: sessionToken,
    ephemeral_key: ephemeralKey,
  };
}

// ---------- App Endpoints (100% AES-256-GCM + HMAC-SHA256 + Anti-Replay + Lớp 4 X-QT-Proof) ----------
async function handleEncryptedAppEndpoint(req, env, processor) {
  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);
  let body, nonce, ts;
  try {
    const rawReq = await req.json();
    nonce = rawReq.nonce;
    ts = rawReq.ts;
    body = await decryptClientPayload(rawReq, env, SERVER_MASTER_SECRET);
  } catch (err) {
    await logSecurityAlert(env, req, "Lỗi giải mã / chữ ký / replay: " + err.message);
    return json({ valid: false, error: "Lỗi xác thực bảo mật: " + err.message }, 403);
  }
  const respData = await processor(body, nonce, ts);
  const encResp = await encryptServerResponse(respData, nonce, ts, SERVER_MASTER_SECRET);
  return json(encResp);
}

// /activate: Kích hoạt key, cấp session_token + ephemeral_key Lớp 4 + asset CDN cơ bản
async function handleActivate(req, env) {
  return handleEncryptedAppEndpoint(req, env, async (body) => {

      created_at: ev.created_at,
      expires_at: ev.expires_at,
      session_token: ev.session_token,
      ephemeral_key: ev.ephemeral_key,
      assets: REMOTE_PUBLIC_ASSETS,
      message: "OK",
    };
  });
}

// /feature-auth: GỌI MỖI KHI DÙNG CHỨC NĂNG — Kiểm tra Lớp 4 X-QT-Proof + Key trực tiếp rồi chỉ trả đúng chức năng đó!
async function handleFeatureAuth(req, env) {
  return handleEncryptedAppEndpoint(req, env, async (body, nonce, ts) => {
    const feature = String(body.feature || "").trim().toLowerCase();
    if (!VALID_FEATURES.has(feature)) {
      return { valid: false, error: "Chức năng yêu cầu không hợp lệ." };
    }
    const l4 = await verifyAppLayer4Proof(req, env, body, nonce, ts, SERVER_MASTER_SECRET);
    if (!l4.ok) {
      return { valid: false, cracked: !!l4.cracked, session_expired: !l4.cracked, error: l4.error };
    }

    const ev = await evaluateLicenseForApp(req, env, body, false);
    if (!ev.valid) return ev;

    const scopedConfig = buildScopedFeatureConfig(feature);
    if (!scopedConfig) {
      return { valid: false, error: "Không tìm thấy cấu hình cho chức năng yêu cầu." };
    }

    const ft = await generateFeatureToken(ev.key, ev.machineId, feature, SERVER_MASTER_SECRET);
    return {
      valid: true,
      feature,
      created_at: ev.created_at,
      expires_at: ev.expires_at,
      session_token: ev.session_token,
      ephemeral_key: ev.ephemeral_key,
      feature_token: ft.token,
      feature_exp: ft.exp,
      config: scopedConfig,
    };
  });
}

// /game-asset: Upload ảnh đại diện Fake Game từ Thư viện Ảnh iPhone lên CDN Cloudflare Worker (có kiểm tra Key + Lớp 4)
async function handleGameAssetUpload(req, env) {
  return handleEncryptedAppEndpoint(req, env, async (body, nonce, ts) => {
    const l4 = await verifyAppLayer4Proof(req, env, body, nonce, ts, SERVER_MASTER_SECRET);
    if (!l4.ok) {
      return { valid: false, cracked: !!l4.cracked, session_expired: !l4.cracked, error: l4.error };
    }
    const ev = await evaluateLicenseForApp(req, env, body, false);
    if (!ev.valid) return ev;

    const rawB64 = String(body.image_b64 || "").trim();
    if (!rawB64 || rawB64.length > 600000) {
      return { valid: false, error: "Ảnh không hợp lệ hoặc vượt quá giới hạn dung lượng (400KB)." };
    }

    const imgId = (await sha256Hex(rawB64)).slice(0, 28);
    const kvKey = `IMG:${imgId}`;
    if (MEM_IMAGES.size > 30) {
      const oldest = MEM_IMAGES.keys().next().value;
      if (oldest) MEM_IMAGES.delete(oldest);
    }
    MEM_IMAGES.set(imgId, rawB64);
    const existing = await sysGetRaw(env, kvKey);
    if (!existing) {
      await sysPutRaw(env, kvKey, rawB64, { bestEffort: true, ttlSec: 2592000 });
    }

    const origin = new URL(req.url).origin;
    return {
      valid: true,
      image_url: `${origin}/img/${imgId}.jpg`,
    };
  });
}

// GET /img/<id>.jpg: Phục vụ ảnh đại diện Game cho Discord Media Proxy
async function handleServeGameImage(req, env, pathname) {
  const match = pathname.match(/^\/img\/([a-f0-9]{16,40})\.(?:jpg|jpeg|png)$/i);
  if (!match) return new Response("Not found", { status: 404 });
  const imgId = match[1].toLowerCase();
  const b64 = MEM_IMAGES.get(imgId) || (await sysGetRaw(env, `IMG:${imgId}`));
  if (!b64) return new Response("Image expired or not found", { status: 404 });
  const bytes = base64ToUint8(b64);
  return new Response(bytes, {
    status: 200,
    headers: {
      "Content-Type": "image/jpeg",
      "Cache-Control": "public, max-age=2592000, immutable",
      "Access-Control-Allow-Origin": "*",
    },
  });
}

// ---------- Router ----------
export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    if (req.method === "OPTIONS") return new Response(null, { headers: CORS });
    try {
      if (req.method === "GET" && url.pathname.startsWith("/img/")) {
        return await handleServeGameImage(req, env, url.pathname);
      }
      switch (url.pathname) {
        case "/":                           return new Response("Quoc Thang Studio — Secure Anti-Crack Server v4", { headers: { "Content-Type": "text/plain; charset=utf-8" } });
        case "/admin":                      return await handleAdminPage(req, env);
        case "/admin/challenge":            return await handleAdminChallenge(req, env);
        case "/admin/auth":                 return await handleAdminAuth(req, env);
        case "/admin/logout":               return await handleAdminLogout(req);
        case "/admin/generate":             return await handleGenerateKey(req, env);
        case "/admin/list":                 return await handleListKeys(req, env);
        case "/admin/info":                 return await handleKeyInfo(req, env);
        case "/admin/extend":               return await handleExtendKey(req, env);
        case "/admin/reset_hwid":           return await handleResetHwid(req, env);
        case "/admin/toggle":               return await handleToggleKey(req, env);
        case "/admin/delete":               return await handleDeleteKey(req, env);
        case "/admin/security/get":         return await handleSecGet(req, env);
        case "/admin/security/save":        return await handleSecSave(req, env);
        case "/admin/security/clear_alerts":return await handleSecClearAlerts(req, env);
        case "/activate":                   return await handleActivate(req, env);
        case "/validate":                   return await handleConfig(req, env);
        case "/config":                     return await handleConfig(req, env);
        case "/feature-auth":               return await handleFeatureAuth(req, env);
        case "/game-asset":                 return await handleGameAssetUpload(req, env);
        default:                            return json({ error: "Not found" }, 404);
      }
    } catch (e) {
      return json({ error: "Server error: " + e.message }, 500);
    }
  },
};
