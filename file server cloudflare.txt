// ---------- Helpers Cryptography ----------
const enc = new TextEncoder();
const dec = new TextDecoder();

function base64ToUint8(str) {
  const binary = atob(str);
  const bytes = new Uint8Array(binary.length);
  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i);
  }
  return bytes;
}

function uint8ToBase64(bytes) {
  let binary = "";
  for (let i = 0; i < bytes.byteLength; i++) {
    binary += String.fromCharCode(bytes[i]);
  }
  return btoa(binary);
}

async function getAesKey(seed) {
  const hash = await crypto.subtle.digest("SHA-256", enc.encode(seed));
  return crypto.subtle.importKey("raw", hash, { name: "AES-GCM" }, false, ["encrypt", "decrypt"]);
}

async function hmacSign(msg, secret) {
  const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name: "HMAC", hash: "SHA-256" }, false, ["sign"]);
  const sig = await crypto.subtle.sign("HMAC", key, enc.encode(msg));
  return Array.from(new Uint8Array(sig)).map(b => b.toString(16).padStart(2, "0")).join("");
}

async function decryptClientPayload(reqBody, secret) {
  const { payload, nonce, ts, sig } = reqBody || {};
  if (!payload || !nonce || !ts || !sig) {
    throw new Error("Missing required security envelope");
  }

  // Chống Replay Attack: gói tin phải trong vòng 60 giây
  const nowSec = Math.floor(Date.now() / 1000);
  const reqSec = parseInt(ts);
  if (isNaN(reqSec) || Math.abs(nowSec - reqSec) > 60) {
    throw new Error("Request expired / Replay attack detected");
  }

  // Kiểm tra chữ ký HMAC
  const expectedSig = await hmacSign(`${payload}:${nonce}:${ts}`, secret);
  if (sig.toLowerCase() !== expectedSig.toLowerCase()) {
    throw new Error("Invalid cryptographic signature");
  }

  // Giải mã AES-GCM
  const key = await getAesKey(`${secret}:${nonce}:${ts}`);
  const combined = base64ToUint8(payload);
  if (combined.length < 28) throw new Error("Payload too short");
  const iv = combined.slice(0, 12);
  const ciphertext = combined.slice(12);

  const decrypted = await crypto.subtle.decrypt({ name: "AES-GCM", iv }, key, ciphertext);
  return JSON.parse(dec.decode(decrypted));
}

async function encryptServerResponse(respObj, nonce, ts, secret) {
  const key = await getAesKey(`${secret}:resp:${nonce}:${ts}`);
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const plaintext = enc.encode(JSON.stringify(respObj));
  const encrypted = await crypto.subtle.encrypt({ name: "AES-GCM", iv }, key, plaintext);

  const combined = new Uint8Array(12 + encrypted.byteLength);
  combined.set(iv, 0);
  combined.set(new Uint8Array(encrypted), 12);

  const payload = uint8ToBase64(combined);
  const sig = await hmacSign(`${payload}:${nonce}`, secret);
  return { payload, sig };
}

async function generateSessionToken(key, machineId, expiresAt, secret) {
  const iat = Math.floor(Date.now() / 1000);
  const exp = expiresAt ? Math.floor(expiresAt / 1000) : iat + 86400 * 30;
  const payloadObj = {
    key: key,
    mid: machineId,
    iat: iat,
    exp: exp,
    salt: Math.random().toString(36).substring(2, 10),
  };
  const bodyB64 = uint8ToBase64(enc.encode(JSON.stringify(payloadObj)));
  const sig = await hmacSign(bodyB64, secret);
  return `${bodyB64}.${sig}`;
}

async function handleGenerateKey(req, env) {
  if (req.method !== "POST") return json({ error: "Method not allowed" }, 405);
  const body = await req.json().catch(() => ({}));
  if (!env.ADMIN_SECRET || body.password !== env.ADMIN_SECRET) {
    return json({ error: "Invalid password" }, 401);
  }

  const key = makeKey();
  const days = parseInt(body.days) || 30;
  const expiresAt = nowMs() + days * 86400000;

  const licData = {
    email: body.email || "",
    created_at: nowMs(),
    expires_at: expiresAt,
    machine_id: null,
    disabled: false
  };

  await putLicense(env, key, licData);
  return json({ valid: true, key: key, expires_at: expiresAt });
}

  const lic = await getLicense(env, key);
  if (!lic || lic.disabled) {
    respData = { error: "App has been cracked", cracked: true, valid: false };
  } else if (lic.expires_at && nowMs() > lic.expires_at) {
    respData = { error: "License expired", cracked: true, valid: false };
  } else if (lic.machine_id && lic.machine_id !== machineId) {
    respData = { error: "App has been cracked (hardware mismatch)", cracked: true, valid: false };
  } else {
    const sessionToken = await generateSessionToken(key, machineId, lic.expires_at, DISCORD_CONFIG.app_secret);
    const { app_secret, ...publicCfg } = DISCORD_CONFIG;
    respData = {
      valid: true,
      created_at: lic.created_at || nowMs(),
      expires_at: lic.expires_at,
      session_token: sessionToken,
      config: publicCfg,
    };
  }

  const encResp = await encryptServerResponse(respData, nonce, ts, DISCORD_CONFIG.app_secret);
  return json(encResp);
}

// ---------- Router ----------
export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    if (req.method === "OPTIONS") return new Response(null, { headers: CORS });
    try {
      switch (url.pathname) {
        case "/":               return new Response("Quoc Thang Tools - Secure License Server v2", { headers: { "Content-Type": "text/plain; charset=utf-8" } });
        case "/admin":          return await handleAdminPage();
        case "/admin/generate": return await handleGenerateKey(req, env);
        case "/admin/info":     return await handleKeyInfo(req, env);
        case "/admin/reset_hwid": return await handleResetHwid(req, env);
        case "/admin/toggle":   return await handleToggleKey(req, env);
        case "/activate":       return await handleActivate(req, env);
        case "/validate":       return await handleValidate(req, env);
        case "/config":         return await handleConfig(req, env);
        default:                return json({ error: "Not found" }, 404);
      }
    } catch (e) {
      return json({ error: "Server error: " + e.message }, 500);
    }
  },
};
